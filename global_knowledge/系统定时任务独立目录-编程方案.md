# 系统定时任务独立目录 — 编程方案

**目标项目**：`E:\code\kemo-agent`
**交付对象**：编程智能体

---

## 一、问题

当前三个系统级定时任务（记忆晋升扫描、临时重要记忆巡检、临时重要记忆每日整理）存放在 `users/{user}/task_cron/`，与用户自建定时任务混在一起，仅靠 `system_key` 字段区分。

实际上这三个任务不需要用户上下文数据——它们执行时是"遍历所有用户操作"，不隶属于任何单个用户。

## 二、目标结构

```
cron/
├── task_cron_system/                  ← 新增：系统级定时任务（不区分用户）
│   ├── memory_promotion.json          ← 每 30 秒扫描所有用户的记忆碎片晋升
│   ├── memory_periodic_scan.json      ← 每 N 小时巡检所有用户的临时重要记忆
│   └── memory_daily_consolidate.json  ← 每天定时整理所有用户的临时重要记忆
│
users/{user}/
├── task_cron/                         ← 不变：用户自建定时任务
│   └── cron_xxxx.json
```

## 三、CronStore 改动

### 3.1 `_task_dir()` 支持 system 模式

```python
def _task_dir(root: Path, user: str, *, system: bool = False) -> Path:
    if system:
        return root / "cron" / "task_cron_system"
    return root / "users" / user / "task_cron"
```

```python
class CronStore:
```

### 3.3 校验规则放宽

系统任务不需要 `user` 和 `prompt` 字段（执行时由调度器自动构造），`system_key` 字段退役。

在校验函数 `_validate_task()` 中：

```python
# 系统目录的任务允许 user 为空、prompt 为空
# 但需要增加 "system" 作为合法的 exec_mode 值
EXEC_MODES = frozenset({"agent", "subagent", "function", "system"})
```

系统任务的 JSON 结构：

```json
{
  "task_id": "memory_promotion",
  "title": "记忆碎片到期晋升检查",
  "user": "",
  "type": "recurring",
  "interval_seconds": 30,
  "next_run_at": "2026-07-22T00:00:00+08:00",
  "latest_run_at": "",
  "status": "enabled",
```

`system_key` 字段从 schema 中移除（`_validate_task` 和 `normalize_task` 不再要求该字段）。历史任务迁移时自动丢弃 `system_key`。

### 4.4 `_reconcile_system_task()` 简化

不再需要 `user` 参数，不再写入 `prompt` / `system_key`：

```python
def _reconcile_system_task(
    store: CronStore,
    current: dict[str, Any] | None,
    *,
    spec: dict[str, Any],
    now: datetime,
) -> dict[str, Any]:
    if current is None:
        draft = {
            "type": spec["type"],
            "next_run_at": "",
        }
        if spec["type"] == "recurring":
            draft["interval_seconds"] = spec["interval_seconds"]
        elif spec["type"] == "daily":
            draft["time"] = spec["time"]
        draft["next_run_at"] = compute_next_run(draft, after=now)
        return store.create(
            normalize_task(
                title=spec["title"],
                prompt="",       # 系统任务无需 prompt
                user="",        # 系统任务不属于任何用户
                type=spec["type"],
                interval_seconds=spec.get("interval_seconds"),
                time=spec.get("time"),
                next_run_at=draft["next_run_at"],
                exec_mode=spec["exec_mode"],
                task_id=spec["task_id"],
            )
        )

    schedule_changed = not _same_schedule(current, spec)
    changed = (
        schedule_changed
        or current.get("title") != spec["title"]
        or current.get("exec_mode") != spec["exec_mode"]
    )
    if not changed:
        return current

    def mutate(task: dict[str, Any]) -> dict[str, Any]:
        task.update({
            "title": spec["title"],
            "type": spec["type"],
            "exec_mode": spec["exec_mode"],
        })
        task.pop("interval_seconds", None)
        task.pop("time", None)
        if spec["type"] == "recurring":
            task["interval_seconds"] = spec["interval_seconds"]
        elif spec["type"] == "daily":
            task["time"] = spec["time"]
        if schedule_changed:
            task["next_run_at"] = compute_next_run(task, after=now)
        return task

    return store.update(current["task_id"], mutate)
```

### 4.5 `CronScheduler._scan_user()` 改为两层扫描

```python
def scan_once(self) -> int:
    from run.users import list_users

    executed = 0
    now = datetime.now(BEIJING)

    # 第一层：系统任务（遍历所有用户执行）
    if not self._stop_event.is_set():
        executed += self._scan_system_tasks(now)

    # 第二层：用户任务（per-user）
    for user in list_users(self.root):
        if self._stop_event.is_set():
            break
        try:
            executed += self._scan_user_tasks(user, now)
        except Exception as exc:
            if self.on_error:
                self.on_error(user, exc)
    return executed


def _scan_system_tasks(self, now: datetime) -> int:
    """扫描 cron/task_cron_system/，对到期任务遍历所有用户执行。"""
    from run.users import list_users

    store = CronStore(self.root, "__system__", system=True)
    tasks = store.list_tasks()
    executed = 0

    for task in tasks:
        if self._stop_event.is_set():
            break
        status = task.get("status", "")
        if status not in {"enabled", "failed"}:
            continue
        if status == "enabled" and not is_due(str(task.get("next_run_at") or ""), now=now):
            continue

        task_id = str(task["task_id"])
        action = task.get("action", task_id)

        # 对每个用户执行
        for user in list_users(self.root):
            if self._stop_event.is_set():
                break
            try:
                result = execute_cron_task(
                    root=self.root,
                    user=user,
                    task_id=task_id,
                    provider_factory=self.provider_factory,
                    tool_registry_factory=self.tool_registry_factory,
                    cancel_event=self._stop_event,
                    system_task=task,  # 传入系统任务定义
                )
                executed += 1
                if self.on_task_executed:
                    self.on_task_executed(user, task_id, result)
            except Exception as exc:
                if self.on_error:
                    self.on_error(user, exc)

        # 更新系统任务的 next_run_at
        try:
            def advance(t: dict[str, Any]) -> dict[str, Any]:
                t["next_run_at"] = compute_next_run(t, after=now)
                t["latest_run_at"] = now.astimezone(BEIJING).isoformat()
                return t
            store.update(task_id, advance)
        except Exception:
            pass

    return executed


def _scan_user_tasks(self, user: str, now: datetime) -> int:
    """扫描 users/{user}/task_cron/，执行到期的用户任务。"""
    store = CronStore(self.root, user)
    tasks = store.list_tasks()
    executed = 0

    for task in tasks:
        if self._stop_event.is_set():
            break
        # ... 原有逻辑不变 ...
    return executed
```

## 五、执行器改动

**文件**：`cron/executor.py`

系统任务执行时，`exec_mode: "system"` 的任务需要分发到对应逻辑：

```python
def execute_cron_task(
    *,
    root: Path,
    user: str,
    task_id: str,
    ...,
    system_task: dict[str, Any] | None = None,
) -> dict[str, Any]:
    # 如果是系统任务，走系统执行逻辑
    if system_task is not None:
        action = system_task.get("action", task_id)
        if action == "memory_promotion":
            return _execute_memory_promotion(root, user)
        elif action == "periodic_scan":
            return _execute_periodic_scan(root, user)
        elif action == "daily_consolidate":
            return _execute_daily_consolidate(root, user)

    # 用户任务走原有逻辑
    ...
```

三个系统执行函数复用现有的子代理调用逻辑（`memory_temporary_important` 和 `self_improve`），只是不再从 `prompt` 字段解析，而是直接硬路由。

### 7.1 旧任务清理

启动时检测 `users/{user}/task_cron/` 中是否存在 `system_key` 非空的旧系统任务，自动删除：

```python
def cleanup_old_system_tasks(root: Path, user: str) -> int:
    store = CronStore(root, user)
    deleted = 0
    for task in store.list_tasks():
        if task.get("system_key"):
            store.delete(task["task_id"])
            deleted += 1
    return deleted
```
